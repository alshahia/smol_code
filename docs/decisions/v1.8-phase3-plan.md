## Question

How do we implement decision 0025 sec 6.5 (Dashboard + a11y + power features)
as a sequence of small, verifiable tasks that finish in 3-5 days with the
validation gates from sec 13.4?

## Findings

### A. Scope (per sec 6.5)

- **A6 Dashboard + cost projection (P1)** - top tab with tokens / runs / errors
  + sparkline + per-provider cost rates (Q5 confirmed: hardcoded defaults +
  `Settings.cost_rates` env override)
- **B2 keyboard shortcuts (P1)** - global shortcut router (`Cmd+Enter` = submit,
  `Cmd+.` = stop, `Cmd+K` = focus command palette, `Cmd+/` = help)
- **B3 search across runs (P1)** - RunHistory filter extended to search by
  task text + provider + tier + date range (the Phase 0 filter is task-text only)
- **B4 rerun / retry (P1)** - `POST /api/runs/{id}/retry` (re-use same task +
  settings; new run_id) + `POST /api/runs/{id}/rerun` (re-use task verbatim)
- **B5 export (P2)** - `GET /api/runs/{id}/export` returns JSON download with
  full event log + RunSummary + tokens + subagent_history
- **B6 accessibility (P1)** - axe-core scan + WCAG 2.2 Level AA fixes (color
  contrast, keyboard focus, ARIA labels on all interactive controls)
- **B7 retry on transient failure (P1)** - `/api/runs/{id}/retry` also handles
  the B7 case (transient infra failures retry; user-aborted runs don't)
- **B10 auto-approve banner (P1)** - when a destructive tool has been approved
  with "no more prompts", show a persistent banner with a "Disable" button
  that revokes the auto-approve flag mid-run
- **NEW (Phase 0 sec 14.8 #3 UI consumption):** render `subagent_history: list<SubAgentSummary>`
  in Inspector (Phase 2 put the field on the wire; this phase consumes it)

### B. PREWORK - test infra

Phase 3 sec 6.5 acceptance gate requires Vitest + Testing Library + axe-core.
This is the right phase to add the dependency weight (we now have enough
surface - 4 new components + the existing App.tsx - to make it worth it).

Decision rationale (decision 0025 sec 6.6): add Vitest + axe-core together
with Playwright smoke; do not over-invest in FE test infra prematurely.

## Decision

Phase 3 ships in this order: PREWORK (test infra) -> BE TDD -> FE TDD ->
validation gates -> commit + push. Each task is small enough to verify in
isolation. Test-driven: write the test first (RED), implement (GREEN),
refactor (REFACTOR). Use the `test-driven-development` skill per AGENTS.md.

## Code Impact

Per sec 6.5 LOC table: BE ~270 LOC, FE ~585 LOC, tests ~320 LOC = **~1175 LOC net**
(realistic drift: +2000-2500 net including TDD scaffolding + Vitest config).

@app.post('/api/runs/{run_id}/retry', response_model=RunStartResponse)
def retry_run(run_id: str, settings: Settings = Depends(get_settings)) -> RunStartResponse:
    mgr = get_run_manager()
    run = mgr.get(run_id)  # KeyError -> 404
    if run.status not in (STATUS_DONE, STATUS_ERROR, STATUS_STOPPED):
        raise HTTPException(409, 'Can only retry a completed/failed/stopped run')
    return mgr.start_or_enqueue_run(
        task=run.task, tier=run.tier, project=run.project, session_id=run.session_id,
        provider_override=run.provider_override, model_override=run.model_override,
        api_key_value=run.api_key_value, parent_retry_of=run.id,
    )

@app.post('/api/runs/{run_id}/rerun', response_model=RunStartResponse)
def rerun_run(run_id: str, settings: Settings = Depends(get_settings)) -> RunStartResponse:
    mgr = get_run_manager()
    run = mgr.get(run_id)
    if run.status != STATUS_DONE:
        raise HTTPException(409, 'Can only rerun a completed run')
    return mgr.start_or_enqueue_run(
        task=run.task, tier=run.tier, project=run.project, session_id=run.session_id,
        provider_override=run.provider_override, model_override=run.model_override,
        api_key_value=run.api_key_value, parent_rerun_of=run.id,
    )

@app.get('/api/runs/{run_id}/export')
def export_run(run_id: str, settings: Settings = Depends(get_settings)) -> Response:
    mgr = get_run_manager()
    run = mgr.get(run_id)
    events = mgr.events(run_id)  # snapshot
    payload = ExportPayload(
        summary=run.summary_dict(),
        events=events,
        subagent_history=run.subagent_history,
        exported_at=time.time(),
        schema_version=1,
    )
    return Response(
        content=payload.model_dump_json(indent=2),
        media_type='application/json',
        headers={'Content-Disposition': f'attachment; filename="run-{run_id}.json"'},
    )

# smolcode/src/smolcode/web/dashboard.py
from collections import Counter
from dataclasses import dataclass, field
import time
from smolcode.model_catalog import cost_for
from smolcode.config import Settings

@dataclass
class DashboardResponse:
    runs_today: int = 0
    tokens_today: TokenSummary = field(default_factory=lambda: TokenSummary(0, 0, 0))
    errors_today: int = 0
    by_provider: dict[str, TokenSummary] = field(default_factory=dict)
    sparkline: list[int] = field(default_factory=list)  # 24 buckets, last 24h
    cost_estimate_usd_today: float = 0.0
    generated_at: float = field(default_factory=time.time)

def compute_dashboard(mgr, audit, settings: Settings) -> DashboardResponse:
    runs = mgr.list_all_runs()  # returns list of Run dataclasses
    now = time.time()
    day_start = now - 86400
    today = [r for r in runs if r.started_at and r.started_at >= day_start]
    tokens = TokenSummary(
        input=sum(r.tokens.input for r in today),
        output=sum(r.tokens.output for r in today),
        total=sum(r.tokens.total for r in today),
    )
    by_provider = Counter(r.provider for r in today if r.provider)
    by_provider_tokens = {
        p: TokenSummary(
            input=sum(r.tokens.input for r in today if r.provider == p),
            output=sum(r.tokens.output for r in today if r.provider == p),
            total=sum(r.tokens.total for r in today if r.provider == p),
        )
        for p in by_provider
    }
    sparkline = _sparkline_24h(today, now)
    cost = sum(
        cost_for(r.provider, r.model, r.tokens.input, r.tokens.output, cache_hit=r.tokens.cache_hit or 0, settings=settings)
        for r in today
    )
    return DashboardResponse(
        runs_today=len(today),
        tokens_today=tokens,
        errors_today=audit.count_since(day_start, level='ERROR'),
        by_provider=by_provider_tokens,
        sparkline=sparkline,
        cost_estimate_usd_today=cost,
    )

def _sparkline_24h(runs, now: float) -> list[int]:
    """24 buckets, last 24h, oldest first."""
    buckets = [0] * 24
    for r in runs:
        if not r.started_at:
            continue
        age_h = (now - r.started_at) / 3600
        bucket = int(23 - age_h)  # 0 = oldest, 23 = current hour
        if 0 <= bucket < 24:
            buckets[bucket] += r.tokens.total or 0
    return buckets
```

# model_catalog.py - additions
from smolcode.config import Settings, ConfigError

DEFAULT_COST_RATES: dict[str, dict[str, tuple[float, float, float]]] = {
    # provider -> model -> (input_per_1k, output_per_1k, cache_per_1k)
    'openai': {
        'gpt-4o': (0.005, 0.015, 0.0),
        'gpt-4o-mini': (0.00015, 0.0006, 0.0),
        'o1-preview': (0.015, 0.06, 0.0),
    },
    'anthropic': {
        'claude-3-5-sonnet-latest': (0.003, 0.015, 0.0),
        'claude-3-5-haiku-latest': (0.0008, 0.004, 0.0),
    },
    'minimax': {
        'MiniMax-M3': (0.001, 0.002, 0.0),
    },
}

def cost_for(
    provider: str | None, model: str | None,
    input_tokens: int, output_tokens: int, cache_hit: int = 0,
    settings: Settings | None = None,
) -> float:
    if not provider or not model:
        return 0.0
    rates = _resolve_rates(provider, model, settings)
    if rates is None:
        return 0.0  # graceful degradation
    in_rate, out_rate, cache_rate = rates
    cost = (input_tokens / 1000.0) * in_rate + (output_tokens / 1000.0) * out_rate
    if cache_hit:
        cost += (cache_hit / 1000.0) * cache_rate
    return round(cost, 6)

def _resolve_rates(provider, model, settings):
    override = settings.cost_rates if settings else None
    if override and provider in override and model in override[provider]:
        return tuple(override[provider][model])
    if provider in DEFAULT_COST_RATES and model in DEFAULT_COST_RATES[provider]:
        return DEFAULT_COST_RATES[provider][model]
    return None
```

    raw = os.environ.get('SMOLCODE_COST_RATES', '').strip()
    if not raw:
        return {}
    try:
        parsed = json.loads(raw)
    except json.JSONDecodeError as e:

    # validate shape: {provider: {model: [in, out, cache]}}
    for prov, models in parsed.items():
        if not isinstance(models, dict):

    return parsed

def export_event_log(run: Run) -> list[StreamEvent]:
    """Snapshot the SSE event log for a run. Truncates observations over 8KB."""
    events = []
    for raw in run.events_snapshot(max_events=2000):  # bounded
        ev = StreamEvent.model_validate(raw)
        # truncate long strings in observations
        if ev.type == 'step.action' and ev.observations:
            for o in ev.observations:
                if isinstance(o, str) and len(o) > 8192:
                    o = o[:8192] + '...[truncated]...'
        events.append(ev)
    return events
```

export type Shortcut = 'submit' | 'stop' | 'palette' | 'help';

const isMac = navigator.platform.toUpperCase().includes('MAC');

export function installKeyboardRouter(handlers: Record<Shortcut, () => void>) {
  function onKey(e: KeyboardEvent) {
    const meta = isMac ? e.metaKey : e.ctrlKey;
    if (!meta) return;
    const inEditable = ['INPUT', 'TEXTAREA'].includes((e.target as HTMLElement)?.tagName ?? '');
    if (e.key === 'Enter' && !e.shiftKey) {
      if (!inEditable) handlers.submit();
      e.preventDefault();
    } else if (e.key === '.') {
      handlers.stop();
      e.preventDefault();
    } else if (e.key.toLowerCase() === 'k') {
      handlers.palette();
      e.preventDefault();
    } else if (e.key === '/') {
      handlers.help();
      e.preventDefault();
    }
  }
  window.addEventListener('keydown', onKey);
  return () => window.removeEventListener('keydown', onKey);
}
```
